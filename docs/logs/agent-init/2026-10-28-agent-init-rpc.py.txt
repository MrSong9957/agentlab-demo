import json
import pathlib
import queue
import subprocess
import threading

process = subprocess.Popen(['codex', 'app-server', '--listen', 'stdio://'], stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
messages = queue.Queue()
errors = []

def read_stdout():
    for line in process.stdout:
        try:
            messages.put(json.loads(line))
        except json.JSONDecodeError:
            messages.put({'unparsed': line[:300]})

def read_stderr():
    for line in process.stderr:
        errors.append(line)

threading.Thread(target=read_stdout, daemon=True).start()
threading.Thread(target=read_stderr, daemon=True).start()
sequence = 0

def request(method, params):
    global sequence
    sequence += 1
    process.stdin.write(json.dumps({'id': sequence, 'method': method, 'params': params}) + '\n')
    process.stdin.flush()
    while True:
        message = messages.get(timeout=50)
        if message.get('id') != sequence:
            continue
        if 'error' in message:
            raise RuntimeError(json.dumps(message['error'], ensure_ascii=False))
        return message['result']

try:
    result = request('initialize', {'clientInfo': {'name': 'agent-init-check', 'version': '1'}, 'capabilities': {'experimentalApi': True}})
    print('initialize: ok', flush=True)
    process.stdin.write(json.dumps({'method': 'initialized'}) + '\n')
    process.stdin.flush()
    listed = request('skills/list', {'cwds': ['/workspace'], 'forceReload': True})
    skills = []
    for item in listed['data']:
        print('skill errors:', json.dumps(item['errors'], ensure_ascii=False), flush=True)
        assert not item['errors'], item['errors']
        skills.extend(item['skills'])
    selected = [s for s in skills if s['name'].startswith('ce-') or s['name'] in ['lfg', 'wtf', 'last30days', 'tdd', 'comprehensive-testing', 'code-review', 'diagnosing-bugs']]
    print('selected skills:', json.dumps([{k:s.get(k) for k in ['name','enabled','scope','path','pluginId']} for s in selected], ensure_ascii=False, indent=2), flush=True)
    expected = {'last30days', 'tdd', 'comprehensive-testing', 'code-review', 'diagnosing-bugs'}
    enabled = {s['name'] for s in skills if s['enabled']}
    print('required skills enabled:', sorted(expected & enabled), flush=True)
    assert expected <= enabled, expected - enabled
    print('standalone skill count:', len(skills), flush=True)
    plugin = request('plugin/read', {'pluginName': 'compound-engineering', 'marketplacePath': '/home/dev/.codex/.tmp/marketplaces/compound-engineering-plugin/.agents/plugins/marketplace.json'})['plugin']
    print('CE plugin:', json.dumps({k:plugin['summary'].get(k) for k in ['id','enabled','installed','localVersion','version']}, ensure_ascii=False), flush=True)
    print('CE skills:', json.dumps([{k:s.get(k) for k in ['name','enabled','path']} for s in plugin['skills']], ensure_ascii=False, indent=2), flush=True)
    assert plugin['summary']['enabled'] and plugin['summary']['installed']
    ce_required = {'compound-engineering:' + name for name in ['ce-setup', 'ce-plan', 'ce-work', 'ce-debug', 'ce-code-review', 'ce-compound']}
    ce_enabled = {s['name'] for s in plugin['skills'] if s['enabled']}
    print('CE skills count:', len(plugin['skills']), flush=True)
    assert ce_required <= ce_enabled, ce_required - ce_enabled
    started = request('thread/start', {'cwd': '/workspace', 'ephemeral': True})
    sources = started.get('instructionSources', [])
    print('fresh ephemeral session instructionSources:', json.dumps(sources, ensure_ascii=False), flush=True)
    assert '/home/dev/.codex/AGENTS.md' in sources, sources
    assert '/workspace/AGENTS.md' in sources, sources
    print('fresh session rule loading: PASS', flush=True)
    print('No turn/start request was sent; no model inference or delegated task was started.', flush=True)
finally:
    process.terminate()
    try:
        process.wait(timeout=5)
    except subprocess.TimeoutExpired:
        process.kill()
        process.wait()
    if errors:
        print('app-server stderr:', ''.join(errors)[-5000:], flush=True)
